  <script>
    (function () {
      var BUILD = {buildInfoJson};

      var ICON_DOWNLOAD = "M5 20h14v-2H5v2zm7-18L5.33 9h4.17v6h5V9h4.17L12 2z";
      var ICON_PLAY = "M8 5v14l11-7z";

      var target = BUILD.buildTarget || "";
      var kind = target === "Android" ? "Android"
               : target.indexOf("StandaloneWindows") === 0 ? "Windows"
               : target === "WebGL" ? "WebGL"
               : "";

      document.body.setAttribute("data-target", kind);

      var el = function (id) { return document.getElementById(id); };
      var btn = el("primary");

      // ------------------------------------------------------------------
      // Header
      // ------------------------------------------------------------------

      el("title").textContent = BUILD.gameName;
      el("version").textContent = [BUILD.channel, BUILD.version, BUILD.platform || kind || target]
        .filter(function (x) { return !!x; })
        .join(" · ");

      if (BUILD.iconUrl) {
        var img = document.createElement("img");
        img.src = BUILD.iconUrl;
        img.alt = "";
        img.onload = function () { el("icon").replaceChildren(img); };
      }

      // ------------------------------------------------------------------
      // Primary action
      // ------------------------------------------------------------------

      function setPrimary(label, href, iconPath, isDownload) {
        btn.hidden = false;
        btn.href = href;
        el("label").textContent = label;
        el("primary-icon").setAttribute("d", iconPath);

        if (isDownload) {
          el("primary-icon").setAttribute("transform", "rotate(180 12 11)");
          btn.setAttribute("download", decodeURIComponent(href.split("/").pop()));
        }
      }

      if (!BUILD.primaryUrl) {
        el("blurb").textContent = "There's no direct download for this platform. Open More below for the build files.";
      } else if (kind === "Android") {
        var isAndroid = /android/i.test(navigator.userAgent);

        if (isAndroid) {
          // Chrome Intent URL: launches real Chrome (not an in-app browser) on the APK,
          // which then downloads it and offers to install.
          var withoutScheme = BUILD.primaryUrl.replace(/^https?:\/\//, "");
          setPrimary("Download now with Chrome", "intent://" + withoutScheme + "#Intent;scheme=https;package=com.android.chrome;end", ICON_DOWNLOAD, false);
          el("blurb").textContent = "Tap below to open this build in Chrome and install it on your device.";

          var direct = document.createElement("a");
          direct.href = BUILD.primaryUrl;
          direct.textContent = "Download directly";
          el("hint").append("Chrome not installed? ", direct, ".");
        } else {
          setPrimary("Download APK", BUILD.primaryUrl, ICON_DOWNLOAD, true);
          el("blurb").textContent = "Download the Android build to your computer, then copy it to your device to install.";
        }
      } else if (kind === "Windows") {
        setPrimary("Download for Windows", BUILD.primaryUrl, ICON_DOWNLOAD, true);
        el("blurb").textContent = "Download the zip, extract it, and run the game.";
      } else if (kind === "WebGL") {
        setPrimary("Play now", BUILD.primaryUrl, ICON_PLAY, false);
        el("blurb").textContent = "Runs right in your browser.";
      } else {
        setPrimary("Download", BUILD.primaryUrl, ICON_DOWNLOAD, true);
      }

      // ------------------------------------------------------------------
      // More
      // ------------------------------------------------------------------

      function formatDate(value) {
        var date = new Date(value);
        return isNaN(date.getTime()) || date.getFullYear() < 2000 ? "—" : date.toLocaleString();
      }

      function formatDuration(seconds) {
        if (!(seconds > 0)) return "—";
        var h = Math.floor(seconds / 3600);
        var m = Math.floor((seconds % 3600) / 60);
        var s = Math.round(seconds % 60);
        return (h ? h + "h " : "") + (h || m ? m + "m " : "") + s + "s";
      }

      function formatSize(bytes) {
        if (!(bytes > 0)) return "";
        var units = ["B", "KB", "MB", "GB"];
        var i = Math.min(units.length - 1, Math.floor(Math.log(bytes) / Math.log(1024)));
        return (bytes / Math.pow(1024, i)).toFixed(i ? 1 : 0) + " " + units[i];
      }

      var rows = [
        ["Game", BUILD.gameName],
        ["Game ID", BUILD.gameId],
        ["Channel", BUILD.channel],
        ["Platform", BUILD.platform],
        ["Build Target", BUILD.buildTarget],
        ["Environment", BUILD.environment],
        ["Version", BUILD.version],
        ["Unity", BUILD.unityVersion],
        ["Started", formatDate(BUILD.buildStartTime)],
        ["Finished", formatDate(BUILD.buildEndTime)],
        ["Duration", formatDuration(BUILD.totalTimeSeconds)],
        ["Size", formatSize(BUILD.totalSize) || "—"],
        ["Warnings", String(BUILD.totalWarnings)],
        ["Errors", String(BUILD.totalErrors)],
        ["Build ID", BUILD.guid, true]
      ];

      rows.forEach(function (row) {
        if (!row[1]) return;
        var tr = document.createElement("tr");
        var name = document.createElement("td");
        var value = document.createElement("td");
        name.textContent = row[0];

        if (row[2]) {
          var code = document.createElement("code");
          code.textContent = row[1];
          value.appendChild(code);
        } else {
          value.textContent = row[1];
        }

        tr.append(name, value);
        el("info").appendChild(tr);
      });

      function addFiles(list, files) {
        (files || []).forEach(function (file) {
          var li = document.createElement("li");
          var a = document.createElement("a");
          var size = document.createElement("span");
          a.href = file.url;
          a.textContent = file.path;
          size.textContent = formatSize(file.size);
          li.append(a, size);
          list.appendChild(li);
        });
      }

      addFiles(el("artifacts"), BUILD.artifacts);
      addFiles(el("output"), BUILD.outputFiles);

      var outputCount = (BUILD.outputFiles || []).length;
      el("output-summary").textContent = outputCount + (outputCount === 1 ? " file" : " files");
    })();
  </script>
