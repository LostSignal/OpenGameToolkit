  <script>
    (function () {
      // The only thing you should need to change between builds.
      var APK_URL = "{apkUrl}";

      var isAndroid = /android/i.test(navigator.userAgent);

      var btn   = document.getElementById("download");
      var label = document.getElementById("label");
      var blurb = document.getElementById("blurb");
      var hint  = document.getElementById("hint");

      var fileName = decodeURIComponent(APK_URL.split("/").pop());
      document.getElementById("filename").textContent = fileName;

      if (isAndroid) {
        // Chrome Intent URL: launches real Chrome (not an in-app browser) on the APK,
        // which then downloads it and offers to install.
        var withoutScheme = APK_URL.replace(/^https?:\/\//, "");
        btn.href = "intent://" + withoutScheme + "#Intent;scheme=https;package=com.android.chrome;end";
        label.textContent = "Download in Chrome";
        blurb.textContent = "Tap below to open this build in Chrome and install it on your device.";
        hint.innerHTML = 'Chrome not installed? <a href="' + APK_URL + '">Download directly</a>.';
      } else {
        // Desktop / everything else: plain download straight to the Downloads folder.
        btn.href = APK_URL;
        btn.setAttribute("download", fileName);
        label.textContent = "Download APK";
        blurb.textContent = "Download the Android build to your computer, then copy it to your device to install.";
        hint.textContent = "";
      }
    })();
  </script>
